#include <DHT.h>

#define DHTPIN 2       // Pin del sensor DHT11
#define DHTTYPE DHT11  // Tipo de sensor
#define LED_PIN 13     // Pin del LED (puedes usar el LED integrado de la placa)

DHT dht(DHTPIN, DHTTYPE);

String comandoRecibido = "";
bool ledEncendido = false;

void setup() {
  Serial.begin(9600);
  dht.begin();
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW); // LED apagado al iniciar
  Serial.println("Sistema iniciado");
}

void loop() {
  // 1. Leer y enviar datos del sensor cada 2 segundos
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();

  if (!isnan(temp) && !isnan(hum)) {
    Serial.print("Temperatura: ");
    Serial.print(temp, 2);
    Serial.print(" °C y Humedad: ");
    Serial.print(hum, 0);
    Serial.println("%");
  }

  // 2. Revisar si hay comandos desde Node.js
  if (Serial.available() > 0) {
    comandoRecibido = Serial.readStringUntil('\n');
    comandoRecibido.trim();

    if (comandoRecibido == "LIGHT_ON") {
      digitalWrite(LED_PIN, HIGH);
      ledEncendido = true;
      Serial.println("LED_ENCENDIDO");
    } 
    else if (comandoRecibido == "LIGHT_OFF") {
      digitalWrite(LED_PIN, LOW);
      ledEncendido = false;
      Serial.println("LED_APAGADO");
    }
    else if (comandoRecibido == "STATUS") {
      if (ledEncendido) {
        Serial.println("LED_ENCENDIDO");
      } else {
        Serial.println("LED_APAGADO");
      }
    }
  }

  delay(2000);
}
